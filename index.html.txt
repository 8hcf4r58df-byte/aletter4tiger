      start() {
        if (this.isPlaying) return;
        const AudioContext = window.AudioContext || window.webkitAudioContext;
        if (!AudioContext) return;

        if (!this.ctx) {
          this.ctx = new AudioContext();
        }
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }

        this.isPlaying = true;
        this.isMuted = false;
        this.playNote();
        this.intervalId = setInterval(() => this.playNote(), 1150);
      }

      playNote() {
        if (!this.ctx || this.isMuted) return;

        const now = this.ctx.currentTime;
        const freq = this.notes[this.step % this.notes.length];
        this.step++;

        // Primary soft bell tone
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, now);

        // Subtle warm harmonic overtone
        const overtone = this.ctx.createOscillator();
        const overtoneGain = this.ctx.createGain();
        overtone.type = 'triangle';
        overtone.frequency.setValueAtTime(freq * 2, now);

        gain.gain.setValueAtTime(0.0001, now);
        gain.gain.exponentialRampToValueAtTime(0.045, now + 0.08);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + 2.6);

        overtoneGain.gain.setValueAtTime(0.0001, now);
        overtoneGain.gain.exponentialRampToValueAtTime(0.008, now + 0.06);
        overtoneGain.gain.exponentialRampToValueAtTime(0.0001, now + 1.4);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        overtone.connect(overtoneGain);
        overtoneGain.connect(this.ctx.destination);

        osc.start(now);
        osc.stop(now + 2.7);
        overtone.start(now);
        overtone.stop(now + 1.5);
      }

      toggleMute() {
        if (!this.isPlaying) {
          this.start();
          return false;
        }
        this.isMuted = !this.isMuted;
        if (!this.isMuted && this.ctx && this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
        return this.isMuted;
      }
    }

    const musicBox = new AmbientMusicBox();

    /* ==========================================================================
       2. FLOATING PETALS & CHAMPAGNE SPARKLES CANVAS
       ========================================================================== */
    const canvas = document.getElementById('ambient-canvas');
    const ctx = canvas.getContext('2d');
    let width, height, particles = [];

    function resizeCanvas() {
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
    }

    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    class Particle {
      constructor() {
        this.reset(true);
      }

      reset(initial = false) {
        this.x = Math.random() * width;
        this.y = initial ? Math.random() * height : -20;
        this.type = Math.random() > 0.42 ? 'sparkle' : 'petal';
        this.size = this.type === 'sparkle' ? Math.random() * 2.2 + 1 : Math.random() * 5 + 3.5;
        this.speedY = Math.random() * 0.45 + 0.18;
        this.speedX = (Math.random() - 0.5) * 0.35;
        this.angle = Math.random() * Math.PI * 2;
        this.spin = (Math.random() - 0.5) * 0.015;
        this.opacity = Math.random() * 0.45 + 0.2;
        this.twinkleSpeed = Math.random() * 0.03 + 0.01;
      }

      update() {
        this.y += this.speedY;
        this.x += this.speedX + Math.sin(this.angle) * 0.22;
        this.angle += this.spin;

        if (this.type === 'sparkle') {
          this.opacity += Math.sin(Date.now() * this.twinkleSpeed) * 0.008;
        }

        if (this.y > height + 20 || this.x < -20 || this.x > width + 20) {
          this.reset(false);
        }
      }

      draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate(this.angle);
        ctx.globalAlpha = Math.max(0.08, Math.min(0.65, this.opacity));

        if (this.type === 'sparkle') {
          ctx.fillStyle = '#D6B982';
          ctx.beginPath();
          // Four-point soft star sparkle
          const s = this.size;
          ctx.moveTo(0, -s * 1.6);
          ctx.quadraticCurveTo(0, 0, s * 1.6, 0);
          ctx.quadraticCurveTo(0, 0, 0, s * 1.6);
          ctx.quadraticCurveTo(0, 0, -s * 1.6, 0);
          ctx.quadraticCurveTo(0, 0, 0, -s * 1.6);
          ctx.fill();
        } else {
          // Delicate dusty-rose petal
          ctx.fillStyle = '#D8A6A1';
          ctx.beginPath();
          ctx.ellipse(0, 0, this.size, this.size * 0.58, 0, 0, Math.PI * 2);
          ctx.fill();
        }
        ctx.restore();
      }
    }

    const particleCount = Math.min(34, Math.floor(window.innerWidth / 14));
    for (let i = 0; i < particleCount; i++) {
      particles.push(new Particle());
    }

    function animateParticles() {
      ctx.clearRect(0, 0, width, height);
      particles.forEach(p => {
        p.update();
        p.draw();
      });
      requestAnimationFrame(animateParticles);
    }
    animateParticles();

    /* ==========================================================================
       3. INTERACTIVE SCREEN FLOW & ENVELOPE ANIMATION
       ========================================================================== */
    const screenOpening = document.getElementById('screen-opening');
    const screenLetter = document.getElementById('screen-letter');
    const screenClosing = document.getElementById('screen-closing');

    const envelopeTrigger = document.getElementById('envelopeTrigger');
    const envelopeWrapper = document.getElementById('envelopeWrapper');
    const openBtn = document.getElementById('openBtn');
    const toClosingBtn = document.getElementById('toClosingBtn');
    const replayBtn = document.getElementById('replayBtn');

    const audioToggle = document.getElementById('audioToggle');
    const audioLabel = document.getElementById('audioLabel');

    let isTransitioning = false;

    function switchScreen(fromScreen, toScreen, callback) {
      fromScreen.classList.remove('visible');
      setTimeout(() => {
        fromScreen.classList.remove('active');
        toScreen.classList.add('active');
        window.scrollTo({ top: 0, behavior: 'smooth' });
        requestAnimationFrame(() => {
          requestAnimationFrame(() => {
            toScreen.classList.add('visible');
            if (callback) callback();
          });
        });
      }, 650);
    }

    function revealLetterUnfolding() {
      const paragraphs = document.querySelectorAll('.letter-paragraph');
      const signoff = document.getElementById('letterSignoff');
      const actions = document.getElementById('letterActions');

      // Reset states in case of replay
      paragraphs.forEach(p => p.classList.remove('reveal'));
      signoff.classList.remove('reveal');
      actions.classList.remove('reveal');

      paragraphs.forEach((p, index) => {
        setTimeout(() => {
          p.classList.add('reveal');
        }, 250 + index * 420);
      });

      const totalDelay = 250 + paragraphs.length * 420;
      setTimeout(() => {
        signoff.classList.add('reveal');
      }, totalDelay + 200);

      setTimeout(() => {
        actions.classList.add('reveal');
      }, totalDelay + 650);
    }

    function handleOpenSurprise() {
      if (isTransitioning) return;
      isTransitioning = true;

      // Start gentle background music on first interaction
      musicBox.start();
      audioToggle.classList.add('visible', 'playing');
      audioLabel.textContent = 'Music On';

      // Trigger envelope flap & letter peek animation
      envelopeWrapper.classList.add('opening');

      setTimeout(() => {
        switchScreen(screenOpening, screenLetter, () => {
          revealLetterUnfolding();
          isTransitioning = false;
        });
      }, 1150);
    }

    openBtn.addEventListener('click', handleOpenSurprise);
    envelopeTrigger.addEventListener('click', handleOpenSurprise);
    envelopeTrigger.addEventListener('keydown', (e) => {
      if (e.key === 'Enter' || e.key === ' ') {
        e.preventDefault();
        handleOpenSurprise();
      }
    });

    toClosingBtn.addEventListener('click', () => {
      switchScreen(screenLetter, screenClosing);
    });

    replayBtn.addEventListener('click', () => {
      envelopeWrapper.classList.remove('opening');
      switchScreen(screenClosing, screenOpening);
    });

    audioToggle.addEventListener('click', () => {
      const muted = musicBox.toggleMute();
      if (muted) {
        audioToggle.classList.remove('playing');
        audioLabel.textContent = 'Muted';
      } else {
        audioToggle.classList.add('playing');
        audioLabel.textContent = 'Music On';
      }
    });
  </script>
